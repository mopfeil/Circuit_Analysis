%----- Falstad examples ----------------------------------------------------
% links/<chapter>.tex are generated by "make falstad" (falstad/build_all.py).
% They contain one \falstaddef{name}{chapter}{url} per example circuit.
\makeatletter
\newcommand{\falstaddef}[3]{%
  \expandafter\edef\csname falstad@url@#1\endcsname{\detokenize{#3}}%
  \expandafter\def\csname falstad@dir@#1\endcsname{#2}}
\@for\falstad@ch:={basics,sources,references,bridges,opamps,filters,samplehold,adc,capstone}\do{%
  \IfFileExists{links/\falstad@ch.tex}{\input{links/\falstad@ch.tex}}{}}
% \falstad{name}: clickable reference, opens the circuit in the browser
\newcommand{\falstad}[1]{%
  % allow a ragged line break in front of the (unbreakable) box
  \ifhmode\unskip\hfil\penalty0\hskip 0pt plus -1fil\relax\ \fi
  \ifcsname falstad@url@#1\endcsname
    \fcolorbox{rwucyan}{rwucyan40!35}{\scriptsize\sffamily
      \raisebox{-0.2ex}{\textcolor{rwucyandark}{$\blacktriangleright$}}~Falstad:
      \href{\csname falstad@url@#1\endcsname}{\texttt{examples/\csname falstad@dir@#1\endcsname/\detokenize{#1}.txt}}}%
  \else
    \fbox{\scriptsize Falstad example \detokenize{#1} missing (run make falstad)}%
    \PackageWarning{falstad}{Falstad example #1 not generated}%
  \fi}
\makeatother

%----- exercises and solutions ---------------------------------------------
% \exercise{title}        starts an exercise (numbered per chapter)
% \begin{solution} ... \end{solution}
% Solutions are printed unless \nosolutions is defined (make students).
\newcounter{exercise}[chapter]
\renewcommand{\theexercise}{\thechapter.\arabic{exercise}}
\newcommand{\exercise}[1]{\refstepcounter{exercise}%
  \ifslides\newpage\fi
  \par\medskip\noindent
  {\large\bfseries\color{rwuviolet}Exercise~\theexercise: #1}\par\smallskip
  \setcounter{TeilaufgNr}{1}}
\ifdefined\nosolutions
  \excludeversion{solution}
\else
  \newenvironment{solution}%
    {\ifslides\newpage\fi\par\smallskip\noindent
     {\bfseries\color{rwucyandark}Solution~\theexercise}\par\smallskip
     \setcounter{TeilaufgNr}{1}}%
    {\par\noindent{\color{rwucyan}\rule{\linewidth}{0.3mm}}\par}
\fi
